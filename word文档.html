<script>
let balance = 0;
let currentChip = 10;
let currentBets = {};
let isSpinning = false;
// 欧洲轮盘 37 个数字的标准顺序
const wheelNumbers = [0, 32, 15, 19, 4, 21, 2, 25, 17, 34, 6, 27, 13, 36, 11, 30, 8, 23,
10, 5, 24, 16, 33, 1, 20, 14, 31, 9, 22, 18, 29, 7, 28, 12, 35, 3, 26];
const redNumbers = [1, 3, 5, 7, 9, 12, 14, 16, 18, 19, 21, 23, 25, 27, 30, 32, 34, 36];
function updateDisplay() {
document.getElementById('balance-display').innerText = `RM
${balance.toLocaleString()}`;
document.getElementById('admin-current-balance').innerText = `RM
${balance.toLocaleString()}`;
}
// 初始化数字下注网格
const grid = document.getElementById('number-grid');
for (let i = 1; i <= 36; i++) {
let btn = document.createElement('button');
let isRed = redNumbers.includes(i);
btn.className = `py-2 rounded-xl font-bold text-sm border transition ${isRed ? 'bg-
red-800/80 hover:bg-red-700 border-red-600 text-white' : 'bg-gray-900/80 hover:bg-gray-
800 border-gray-700 text-white'}`;
btn.innerText = i;
btn.onclick = () => placeBet('number', i);
grid.appendChild(btn);
}
function selectChip(val) {
currentChip = val;
document.querySelectorAll('#chip-selector button').forEach(b => {
if(parseInt(b.dataset.val) === val) b.classList.add('selected');
else b.classList.remove('selected');
});
}
function placeBet(type, target) {
if (balance < currentChip) {
alert('余额不足，请先联系管理员充值 RM！');
return;
}
balance -= currentChip;
let key = `${type}_${target}`;
currentBets[key] = (currentBets[key] || 0) + currentChip;
updateDisplay();
document.getElementById('result-text').innerText = `已下注: ${target} (RM
${currentChip})`;
}
function clearBets() {
let refunded = Object.values(currentBets).reduce((a,b)=>a+b, 0);
balance += refunded;
currentBets = {};
updateDisplay();
document.getElementById('result-text').innerText = '已清除所有下注';
}
// Canvas 轮盘绘制
const canvas = document.getElementById('rouletteCanvas');
const ctx = canvas.getContext('2d');
let currentAngle = 0;
function drawWheel(angle = 0) {
ctx.clearRect(0, 0, 320, 320);
ctx.save();
ctx.translate(160, 160);
ctx.rotate(angle);
const sliceAngle = (2 * Math.PI) / wheelNumbers.length;
wheelNumbers.forEach((num, index) => {
ctx.beginPath();
ctx.moveTo(0, 0);
ctx.arc(0, 0, 150, index * sliceAngle, (index + 1) * sliceAngle);
ctx.closePath();
if (num === 0) ctx.fillStyle = '#059669';
else if (redNumbers.includes(num)) ctx.fillStyle = '#dc2626';
else ctx.fillStyle = '#111827';
ctx.fill();
ctx.strokeStyle = '#fbbf24';
ctx.lineWidth = 1;
ctx.stroke();
// 绘制数字
ctx.save();
ctx.rotate(index * sliceAngle + sliceAngle / 2);
ctx.textAlign = 'right';
ctx.fillStyle = '#ffffff';
ctx.font = 'bold 12px Cinzel';
ctx.fillText(num, 135, 4);
ctx.restore();
});
ctx.restore();
}
drawWheel(0);
function spinWheel() {
if (isSpinning) return;
if (Object.keys(currentBets).length === 0) {
alert('请先在盘面上点击下注！');
return;
}
isSpinning = true;
document.getElementById('spin-btn').disabled = true;
document.getElementById('result-text').innerText = '轮盘旋转中...';
let winningIndex = Math.floor(Math.random() * wheelNumbers.length);
let winningNumber = wheelNumbers[winningIndex];
let sliceAngle = (2 * Math.PI) / wheelNumbers.length;
let targetAngle = -(winningIndex * sliceAngle + sliceAngle / 2) + Math.PI * 10; //
多转几圈
let startTime = null;
let duration = 4000; // 4秒
function animate(timestamp) {
if (!startTime) startTime = timestamp;
let progress = (timestamp - startTime) / duration;
if (progress > 1) progress = 1;
// 缓动函数 (Ease Out Cubic)
let easeOut = 1 - Math.pow(1 - progress, 3);
let currentDrawAngle = currentAngle + (targetAngle - currentAngle) * easeOut;
drawWheel(currentDrawAngle);
if (progress < 1) {
requestAnimationFrame(animate);
} else {
isSpinning = false;
currentAngle = targetAngle % (2 * Math.PI);
document.getElementById('spin-btn').disabled = false;
// 结算
let winAmount = 0;
for (let key in currentBets) {
let [type, target] = key.split('_');
if (type === 'number' && parseInt(target) === winningNumber) {
winAmount += currentBets[key] * 36;
}
}
if (winAmount > 0) {
balance += winAmount;
document.getElementById('result-text').innerText = `🎰 恭喜中奖！开出数字:
${winningNumber}，赢得 RM ${winAmount}`;
} else {
document.getElementById('result-text').innerText = `🎰 很遗憾，开出数字:
${winningNumber}`;
}
currentBets = {};
updateDisplay();
}
}
requestAnimationFrame(animate);
}
// 管理员后台逻辑
function openAdminModal() {
document.getElementById('admin-modal').classList.remove('hidden');
document.getElementById('admin-login-view').classList.remove('hidden');
document.getElementById('admin-panel-view').classList.add('hidden');
document.getElementById('admin-pwd').value = '';
}
function closeAdminModal() {
document.getElementById('admin-modal').classList.add('hidden');
}
function verifyAdmin() {
let pwd = document.getElementById('admin-pwd').value;
if (pwd === '8888') {
document.getElementById('admin-login-view').classList.add('hidden');
document.getElementById('admin-panel-view').classList.remove('hidden');
} else {
alert('密码错误！');
}
}
function topupBalance() {
let amt = parseInt(document.getElementById('topup-amount').value);
if (isNaN(amt) || amt <= 0) {
alert('请输入有效的充值金额！');
return;
}
balance += amt;
updateDisplay();
alert(`成功充值 RM ${amt}！`);
closeAdminModal();
}
updateDisplay();
